## Sensitivity (sens_*.csv); atest_fork_s1 / _s2 are wrappers: exec $F/src/atest -S1 "$@" (-S2)
B=test/benchmark/rx_sensitivity.py
python3 $B --atest upstream=$U/src/atest --atest fork=$F/src/atest --gen-packets $U/src/gen_packets --rate 96000 --profiles A+,A,B --conditions flat,deemph6 --ebn0 6:16:1 --frames 300 --seed 301 --jobs 4 --out sens_96k.csv
python3 $B --atest upstream=$U/src/atest --atest fork=$F/src/atest --gen-packets $U/src/gen_packets --rate 192000 --profiles A+,A --conditions flat --ebn0 6:16:1 --frames 300 --seed 302 --jobs 4 --out sens_192k.csv
python3 $B --atest upstream=$U/src/atest --atest fork=$F/src/atest --gen-packets $U/src/gen_packets --rate 88200 --profiles A+ --conditions flat --ebn0 6:16:1 --frames 300 --seed 303 --jobs 4 --out sens_88k.csv
python3 $B --atest upstream=$U/src/atest --atest fork=$F/src/atest --atest fork-S1=atest_fork_s1 --atest fork-S2=atest_fork_s2 --gen-packets $U/src/gen_packets --profiles A+,A --conditions flat,impulse --ebn0 6:14:1 --frames 500 --seed 304 --jobs 4 --out sens_44k.csv --dump-bad bad_44k.txt
for m in 9600 300 2400 4800; do
  python3 $B --mode $m --atest upstream=$U/src/atest --atest fork=$F/src/atest --atest fork-S1=atest_fork_s1 --gen-packets $U/src/gen_packets --conditions flat --ebn0 6:18:1 --frames 300 --seed 305 --jobs 4 --out sens_$m.csv --dump-bad bad_$m.txt
done
